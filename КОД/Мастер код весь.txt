  return phrases.some((phrase) => text.includes(phrase));

  return words.some((word) => new RegExp(`(^|\\s)${word}($|\\s)`).test(text));

function classifyModeration(text) {
  const normalized = normalizeText(text)
    .replace(/[^а-яa-z0-9\s]/g, " ")
    .replace(/\s+/g, " ")
    .trim();

  if (!normalized) {
    return "none";
  }

  const selfDirected = isDirectedAtSelf(normalized);
  const directedAtOther = isDirectedAtOther(normalized);

  if (selfDirected && !directedAtOther) {
    return "none";
  }

  const hasDeletePhrase = containsAnyPhrase(normalized, MODERATION_DELETE_PHRASES);
  const hasWarningPhrase = containsAnyPhrase(normalized, MODERATION_WARNING_PHRASES);
  const hasDeleteWord = containsAnyWord(normalized, MODERATION_DELETE_WORDS);
  const hasWarningWord = containsAnyWord(normalized, MODERATION_WARNING_WORDS);
  const hasHardImperativeAttack = /\b(пиздуйте|съебите|уебывайте)\b/.test(normalized) && /\b(нахуй|на хуй)\b/.test(normalized);
  const hasPluralHardInsult = /\b(мудаки|пидоры|долбоебы|ебаные|ебанутые)\b/.test(normalized);

  if (hasDeletePhrase || hasHardImperativeAttack || (hasPluralHardInsult && (hasDeleteWord || hasHardImperativeAttack)) || (directedAtOther && hasDeleteWord)) {
    return "delete";
  }

  if (hasWarningPhrase || (directedAtOther && hasWarningWord)) {
    return "warn";
  }

  if (hasDeleteWord || hasWarningWord) {
    return "review";
  }

  return "none";
}

function getBillQuestionNumber(text) {
  const normalized = normalizeText(text);
  const standalone = normalized.match(/^\d{1,3}$/);

  if (standalone) {
    const number = Number(standalone[0]);
    if (number >= 1 && number <= 500 && ![111, 222, 333, 444].includes(number)) {
      return number;
    }
  }

  const explicit = normalized.match(/(?:игра[а-я]*|ирга[а-я]*|вопрос[а-я]*)\s*(\d{1,3})/i);
  if (explicit) {
    const number = Number(explicit[1]);
    if (number >= 1 && number <= 500) {
      return number;
    }
  }

  const anyNumber = normalized.match(/\b(\d{1,3})\b/g);
  if (!anyNumber) {
    return null;
  }

  for (const candidate of anyNumber) {
    const number = Number(candidate);
    if (number >= 1 && number <= 500 && ![111, 222, 333, 444].includes(number)) {
      return number;
    }
  }

  return null;
}

function ensureSingleActiveEntry(state) {
  const pendingEntries = state.entries.filter((entry) => entry.status === "pending");

  if (pendingEntries.length === 0) {
    state.entries.forEach((entry) => {
      entry.isActive = false;
    });
    return;
  }

  const alreadyActive = pendingEntries.find((entry) => entry.isActive);
  if (alreadyActive) {
    state.entries.forEach((entry) => {
      if (entry.id !== alreadyActive.id) {
        entry.isActive = false;
      }
    });
    return;
  }

  const firstPending = pendingEntries[0];
  state.entries.forEach((entry) => {
    entry.isActive = entry.id === firstPending.id;
  });
}

function buildQueueText(state) {
  const lines = [
    `<b>${getQueueModeTitle(state.mode)}</b>`,
    state.isOpen ? `<b>${QUEUE_OPEN_LABEL}</b>` : `<b>${QUEUE_CLOSED_LABEL}</b>`,
    `<i>${getQueueInstruction(state.mode)}</i>`,
    "",
    getQueueHeader(),
    "",
  ];

  if (state.entries.length === 0) {
    lines.push("Пока пусто.");
    return lines.join("\n");
  }

  for (const entry of state.entries) {
    let marker = "•";
    if (entry.status === "done") {
      marker = "✅";
    } else if (entry.isActive) {
      marker = "▶";
    }

    lines.push(`${marker} ${entry.author} — ${entry.label}`);
  }

  return lines.join("\n");
}

function addQueueEntryToState(state, entry) {
  state.entries.push(entry);

  if (state.mode === "bill") {
    const priorityOrder = { first: 1, "222": 2, "333": 3, "444": 4 };
    state.entries.sort((a, b) => {
      const diff = (priorityOrder[a.block] ?? 99) - (priorityOrder[b.block] ?? 99);
      if (diff !== 0) {
        return diff;
      }

      return a.createdAt - b.createdAt;
    });
  } else {
    state.entries.sort((a, b) => a.createdAt - b.createdAt);
  }

  ensureSingleActiveEntry(state);
}

function getNextPendingIndex(state, startIndex = 0) {
  for (let i = startIndex; i < state.entries.length; i += 1) {
    if (state.entries[i].status === "pending") {
      return i;
    }
  }

  return -1;
}

function getCurrentActiveIndex(state) {
  return state.entries.findIndex((entry) => entry.isActive && entry.status === "pending");
}

function activateNextPendingFrom(state, index) {
  state.entries.forEach((entry) => {
    entry.isActive = false;
  });

  const nextIndex = getNextPendingIndex(state, index);
  if (nextIndex !== -1) {
    state.entries[nextIndex].isActive = true;
  }
}

function activateFirstPending(state) {
  state.entries.forEach((entry) => {
    entry.isActive = false;
  });

  const firstIndex = getNextPendingIndex(state, 0);
  if (firstIndex !== -1) {
    state.entries[firstIndex].isActive = true;
  }
}

function parseBillQueueEntry(message, state) {
  const rawText = String(message.text || "").trim();
  const normalized = normalizeText(rawText);
  const has111 = /\b111\b/.test(normalized);
  const has222 = /\b222\b/.test(normalized);
  const has333 = /\b333\b/.test(normalized);
  const has444 = /\b444\b/.test(normalized);
  const questionNumber = getBillQuestionNumber(rawText);

  if (questionNumber !== null) {
    if (has111) {
      return makeQueueEntry(message, "first", `111 + игра ${questionNumber}`, rawText);
    }

    return makeQueueEntry(message, "first", `игра ${questionNumber}`, rawText);
  }

  if (has111 || has222 || has333 || has444) {
    return makeQueueEntry(message, "speech", "__speech__", rawText, { kind: "bill_speech" });
  }

  return null;
}

function parseBkQueueEntry(message) {
  const rawText = String(message.text || "").trim();
  const normalized = normalizeText(rawText);

  if (/\b222\b|\b333\b|\b444\b/.test(normalized)) {
    return null;
  }

  const has111 = /\b111\b/.test(normalized);
  const wantsRead = /чит|почитаю/.test(normalized);
  const wantsSpeak = /высказ|выксаз|говор|скажу/.test(normalized);

  if (!has111 && !wantsRead && !wantsSpeak) {
    return null;
  }

  let label = "111";

  if (wantsRead && !wantsSpeak) {
    label = "111 читать";
  } else if (wantsSpeak && !wantsRead) {
    label = "111 высказаться";
  } else if (rawText && rawText !== "111") {
    label = rawText;
  }

  return makeQueueEntry(message, "bk", label, rawText);
}

function parseRsQueueEntry(message) {
  const rawText = String(message.text || "").trim();
  const normalized = normalizeText(rawText);

  if (!/^111(?:[\s!.,?:;#-]*)$/.test(normalized)) {
    return null;
  }

  return makeQueueEntry(message, "rs", "111", rawText);
}

function parseQueueEntry(message, state) {
  if (!state?.isOpen || !state?.mode) {
    return null;
  }

  if (state.mode === "bill") {
    return parseBillQueueEntry(message, state);
  }

  if (state.mode === "bk") {
    return parseBkQueueEntry(message);
  }

  if (state.mode === "rs") {
    return parseRsQueueEntry(message);
  }

  return null;
}

        return Response.json({ ok: true });

          publishQueue: true,
          queueText: buildQueueText(queueState),
          previousMessageId,
        });
      }

      return Response.json({ ok: false, error: "РќРµРёР·РІРµСЃС‚РЅРѕРµ РґРµР№СЃС‚РІРёРµ РѕС‡РµСЂРµРґРё." }, { status: 400 });
    } catch (error) {
      return Response.json({ ok: false, error: error.message }, { status: 400 });
    }

}

function getAuthorLabel(message) {
  const first = message.from?.first_name ?? "";
  const last = message.from?.last_name ?? "";
  const fullName = `${first} ${last}`.trim();
  const username = message.from?.username ? `@${message.from.username}` : "";

  if (fullName && username) {
    return `${fullName} (${username})`;
  }

  if (fullName) {
    return fullName;
  }

  if (username) {
    return username;
  }

  if (message.sender_chat?.title) {
    return `от имени чата: ${message.sender_chat.title}`;
  }

  return "не удалось определить";
}

async function callTelegram(env, method, payload) {
  const response = await fetch(`https://api.telegram.org/bot${env.BOT_TOKEN}/${method}`, {
    method: "POST",
    headers: {
      "content-type": "application/json"
    },
    body: JSON.stringify(payload)
  });

  const data = await response.json();

  if (!data.ok) {
    throw new Error(`${method} failed: ${JSON.stringify(data)}`);
  }

  return data;
}

async function applyQueueResponse(env, result) {
  if (result.publishQueue) {
    const sent = await sendMessage(
      env,
      CHAT_GROUP_ID,
      result.queueText,
      null,
      null,
      null,
      result.parseMode ?? "HTML"
    );

    if (result.previousMessageId) {
      await deleteMessageSafe(env, CHAT_GROUP_ID, result.previousMessageId);
    }

    await callQueueState(env, "set_message_id", {
      messageId: sent.result?.message_id ?? null,
    });
  }

  if (result.closeMessage) {
    await sendMessage(env, CHAT_GROUP_ID, result.closeMessage);
  }

  if (result.cleared && result.previousMessageId) {
    await deleteMessageSafe(env, CHAT_GROUP_ID, result.previousMessageId);
  }
}

async function fetchJsonWithCache(url, cacheName) {
  const now = Date.now();

  if (cacheName === "yozhik" && yozhikCache && now - yozhikCacheTime < DATA_CACHE_TTL_MS) {
    return yozhikCache;
  }

  if (cacheName === "bill" && billCache && now - billCacheTime < DATA_CACHE_TTL_MS) {
    return billCache;
  }

  const response = await fetch(url, {
    headers: {
      accept: "application/json"
    }
  });

  if (!response.ok) {
    throw new Error(`Не удалось загрузить ${cacheName}. HTTP ${response.status}`);
  }

  const data = await response.json();

  if (cacheName === "yozhik") {
    yozhikCache = data;
    yozhikCacheTime = now;
  }

  if (cacheName === "bill") {
    billCache = data;
    billCacheTime = now;
  }

  return data;
}

function getMoscowDateKey() {
  return new Intl.DateTimeFormat("ru-RU", {
    timeZone: "Europe/Moscow",
    day: "2-digit",
    month: "2-digit"
  }).format(new Date());
}

function getMoscowHumanDate() {
  return new Intl.DateTimeFormat("ru-RU", {
    timeZone: "Europe/Moscow",
    day: "numeric",
    month: "long"
  }).format(new Date());
}

function isMoscowTime(hour, minute) {
  const parts = new Intl.DateTimeFormat("en-GB", {
    timeZone: "Europe/Moscow",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
  }).formatToParts(new Date());

  const values = Object.fromEntries(parts.map((part) => [part.type, part.value]));
  return Number(values.hour) === hour && Number(values.minute) === minute;
}

function splitTitleAndBody(text) {
  const normalized = String(text || "").trim();
  const parts = normalized.split(/\n\s*\n/);

  const title = (parts.shift() || "").trim();
  const body = parts.join("\n\n").trim();

  return { title, body };
}

async function sendYozhikToGroup(env) {
  const data = await fetchJsonWithCache(YOZHIK_JSON_URL, "yozhik");

${text}`;

}

async function sendBillToGroup(env, billNumber) {
  if (!Number.isInteger(billNumber) || billNumber < 1 || billNumber > 332) {
    throw new Error("Номер отрывка Билла должен быть от 1 до 332.");
  }

  const data = await fetchJsonWithCache(BILL_JSON_URL, "bill");
  const text = data[String(billNumber)];

  if (!text) {
    throw new Error(`Не нашёл отрывок Билла №${billNumber}`);
  }

  const { title, body } = splitTitleAndBody(text);

  const header = title
    ? `Как это видит Билл. №${billNumber}. ${title}`
    : `Как это видит Билл. №${billNumber}`;

  const messageText = body
    ? `${header}

${body}`
    : `${header}

${text}`;

  await sendMessage(env, CHAT_GROUP_ID, messageText);
}
